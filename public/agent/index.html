将私有工具说明书直接“烘焙”进模型权重

通用模型每次调用工具，必须在 System Prompt 里塞入完整的 JSON Schema。而在你的 Personal Agent 中，核心工具（如读写本地笔记、查 SQLite、执行终端命令、操作网页 DOM）固定在 15~25 个左右。

传统携带 Schema Prompt (每步) ~2,420 Tokens
烘焙后 Zero-Schema Prompt (每步) ~165 Tokens (-93%)
黄金法则：20% 错误自愈轨迹 (Error-Recovery)

合成 500 条多步轨迹时，必须强制注入 20% 的「工具返回报错 -> 模型看懂报错并修正参数重试」样本，防止 3B 小模型在真实环境中遇到报错陷入死循环。

import json
from typing import Literal, Union
from pydantic import BaseModel, Field, ValidationError

# 1. 定义你 Personal Agent 的核心工具严格参数 Schema (用于合成校验，不放进最终训练 Prompt)
class FsReadArgs(BaseModel):
    path: str = Field(..., pattern=r"^(\./|/|[a-zA-Z]:).*")
    max_lines: int = Field(default=100, ge=1, le=500)

class SqlQueryArgs(BaseModel):
    sql: str = Field(..., description="只允许 SELECT 查询语句")

class MemoryPatchArgs(BaseModel):
    section: Literal["user_prefs", "active_tasks", "daily_facts"]
    operation: Literal["append", "replace"]
    content: str

TOOL_REGISTRY = {
    "fs_read": FsReadArgs,
    "sqlite_query": SqlQueryArgs,
    "memory_patch": MemoryPatchArgs,
}

class CompactToolAction(BaseModel):
    """训练 3B 小模型输出的极简无废话 Action 协议 (限制 thought 长度 <= 25 字)"""
    thought: str = Field(..., max_length=40)
    call: Literal["fs_read", "sqlite_query", "memory_patch", "final_answer", "escalate_to_cloud"]
    args: dict

def verify_trajectory(messages: list[dict]) -> tuple[bool, str]:
    """偷师 APIGen 核心：对大模型合成的每一步 Tool Call 进行本地 Pydantic 语法与参数实体验证"""
    for msg in messages:
        if msg["role"] == "assistant":
            try:
                action = CompactToolAction.model_validate_json(msg["content"])
                if action.call in TOOL_REGISTRY:
                    # 校验具体工具的参数是否100%符合你的本地 Python 函数签名
                    TOOL_REGISTRY[action.call].model_validate(action.args)
            except ValidationError as e:
                return False, f"Schema 校验未通过: {e}"
    return True, "OK"

def strip_schema_for_sft(raw_trajectory: list[dict]) -> list[dict]:
    """将冗长的 2400 Token 工具定义 System Prompt 替换为 30 Token 的极简指纹"""
    compact_sys = (
        "You are LocalExecutor-3B. Output strict JSON: "
        '{"thought": "<15字内简短意图>", "call": "", "args": {...}}'
    )
    cleaned = [{"role": "system", "content": compact_sys}]
    for m in raw_trajectory:
        if m["role"] != "system":
            cleaned.append(m)
    return cleaned

# 示例：验证并导出 Unsloth 训练所需的 ShareGPT / ChatML jsonl
if __name__ == "__main__":
    sample_traj = [
        {"role": "system", "content": "...[2400 tokens of verbose JSON schema]..."},
        {"role": "user", "content": "查一下我昨天记在数据库里的开销，并更新到周报记忆里"},
        {"role": "assistant", "content": '{"thought":"先查数据库昨日开销记录","call":"sqlite_query","args":{"sql":"SELECT item, amount FROM expenses WHERE date = date(\'now\', \'-1 day\')"}}'},
        {"role": "tool", "content": '[{"item": "云服务账单", "amount": 42.5}]'},
        {"role": "assistant", "content": '{"thought":"将开销写入active_tasks记忆","call":"memory_patch","args":{"section":"daily_facts","operation":"append","content":"昨日云服务开销 42.5 元"}}'}
    ]
    ok, reason = verify_trajectory(sample_traj)
    if ok:
        sft_ready = strip_schema_for_sft(sample_traj)
        print("校验通过！生成 Zero-Schema 训练样本:\n", json.dumps(sft_ready, ensure_ascii=False, indent=2))

from unsloth import FastLanguageModel
from trl import SFTTrainer
from transformers import TrainingArguments
from datasets import load_dataset
import torch

# 1. 加载 4-bit 量化 Qwen2.5-Coder-3B 基座 (8GB 显存黄金底座)
max_seq_length = 4096
model, tokenizer = FastLanguageModel.from_pretrained(
    model_name="Qwen/Qwen2.5-Coder-3B-Instruct",
    max_seq_length=max_seq_length,
    load_in_4bit=True,          # 开启 4-bit QLoRA
    dtype=None,                 # 自动检测 bf16/fp16
)

# 2. 挂载全线性层 LoRA (r=32 保证能把你的私有工具 API 签名死死背进权重里)
model = FastLanguageModel.get_peft_model(
    model,
    r=32,
    target_modules=["q_proj", "k_proj", "v_proj", "o_proj",
                    "gate_proj", "up_proj", "down_proj"],
    lora_alpha=64,
    lora_dropout=0,             # 设为 0 以启用 Unsloth 底层内核加速
    bias="none",
    use_gradient_checkpointing="unsloth", # 省下 30% 显存，8GB 跑 4096 长度不爆的关键！
    random_state=3407,
)

# 3. 加载你在 Phase 1 洗好的 Zero-Schema 多步工具轨迹数据
dataset = load_dataset("json", data_files="sft_zero_schema_trajectories.jsonl", split="train")

def format_chatml(examples):
    texts = [
        tokenizer.apply_chat_template(msgs, tokenize=False, add_generation_prompt=False)
        for msgs in examples["messages"]
    ]
    return {"text": texts}

dataset = dataset.map(format_chatml, batched=True)

# 4. 8GB VRAM 专属训练超参 (batch_size=1 + grad_accum=8)
trainer = SFTTrainer(
    model=model,
    tokenizer=tokenizer,
    train_dataset=dataset,
    dataset_text_field="text",
    max_seq_length=max_seq_length,
    packing=True,               # 开启样本拼接加速多轮短轨迹训练
    args=TrainingArguments(
        per_device_train_batch_size=1,
        gradient_accumulation_steps=8,
        warmup_steps=15,
        num_train_epochs=3,
        learning_rate=2e-4,
        fp16=not torch.cuda.is_bf16_supported(),
        bf16=torch.cuda.is_bf16_supported(),
        logging_steps=5,
        optim="adamw_8bit",     # 8-bit 优化器进一步压降显存峰值
        weight_decay=0.01,
        lr_scheduler_type="cosine",
        output_dir="outputs/lora_executor",
    ),
)

trainer.train()

# 5. 保存轻量 LoRA 适配器 (用于多 LoRA 热切换) 或直接导出 Q4_K_M GGUF
model.save_pretrained("lora_adapters/lora-tool-executor")
tokenizer.save_pretrained("lora_adapters/lora-tool-executor")
# 若只需单模型部署，也可直接一键合并导出 GGUF:
# model.save_pretrained_gguf("gguf_model", tokenizer, quantization_method="q4_k_m")

PHASE 03 // CONSTRAINED DECODING & GBNF GRAMMAR

                        
偷师 llama.cpp：用 GBNF 状态机物理锁死 0% 格式报错

                    

                    
                        核心收益: JSON 解析成功率 100%
                    
                


                
                

                    

                        

                            底层原理解析
                            Logits Masking 状态机
                        

                        
不要赌 3B 模型不犯错，在采样前直接屏蔽非法 Token

                        

                            微调只能让模型“大概率”输出正确格式，但在长上下文干扰下，3B 模型仍可能漏写引号或编造不存在的工具名。通过编写 GBNF (GGML BNF) 语法规则，推理引擎会在生成每一个 Token 时，将不符合语法树的词表概率直接设为 -∞。

                                {"thought": "查昨日开销", "call": "

                                agent_action.gbnf
                            

                            
                        

                        
# 物理锁死 3B 模型只能输出极简 thought + 合法工具名 + 对应参数结构
root   ::= "{" ws "\"thought\":" ws string "," ws dispatch "}" ws

# 路由分发：一旦生成了特定工具名，后续 args 必须严格匹配该工具的专属语法树！
dispatch ::= fs-read-call | sql-query-call | mem-patch-call | escalate-call | final-call

fs-read-call ::= "\"call\":" ws "\"fs_read\"," ws "\"args\":" ws "{" ws "\"path\":" ws string ("," ws "\"max_lines\":" ws number)? "}"
sql-query-call ::= "\"call\":" ws "\"sqlite_query\"," ws "\"args\":" ws "{" ws "\"sql\":" ws string "}"
mem-patch-call ::= "\"call\":" ws "\"memory_patch\"," ws "\"args\":" ws "{" ws "\"section\":" ws section-enum "," ws "\"operation\":" ws op-enum "," ws "\"content\":" ws string "}"
escalate-call ::= "\"call\":" ws "\"escalate_to_cloud\"," ws "\"args\":" ws "{" ws "\"reason\":" ws string "}"
final-call ::= "\"call\":" ws "\"final_answer\"," ws "\"args\":" ws "{" ws "\"reply\":" ws string "}"

section-enum ::= "\"user_prefs\"" | "\"active_tasks\"" | "\"daily_facts\""
op-enum      ::= "\"append\"" | "\"replace\""

# 基础图元定义
string ::= "\"" ([^"\\] | "\\" (["\\/bfnrt] | "u" [0-9a-fA-F]{4}))* "\""
number ::= [0-9]+
ws     ::= [ \t\n]*

01. Observation Pruning

                        
工具返回值瘦身拦截器

02. Seamless Escalation

                        
动态熔断与云端接力

                        

                            当本地 3B 模型连续两次触发工具报错，或主动调用 escalate_to_cloud 时，无缝将已搜集的本地上下文打包移交云端大模型完成兜底。
                        

                    

                    

                        
03. Nightly Data Flywheel

                        
失败轨迹自动转微调样本

                        

                            每次触发云端兜底时，云端大模型的正确决策会与本地 3B 的失败尝试一起写入 flywheel_dataset.jsonl，周末自动跑 15 分钟增量微调。

import json
import time
from pathlib import Path

class PersonalAgentHarness:
    def __init__(self, local_llm, cloud_llm, event_log_path="main_events.jsonl"):
        self.local_llm = local_llm          # 挂载了 GBNF 与多 LoRA 的本地 llama.cpp 实例
        self.cloud_llm = cloud_llm          # DeepSeek-V3 / Claude 云端兜底客户端
        self.event_log = Path(event_log_path)
        self.flywheel_log = Path("flywheel_correction_samples.jsonl")

    def prune_observation(self, raw_obs: str, max_chars: int = 600) -> str:
        """防止长网页/长日志冲垮 3B 模型的注意力窗口"""
        if len(raw_obs) <= max_chars:
            return raw_obs
        head = raw_obs[:400]
        tail = raw_obs[-150:]
        return f"{head}\n...[已截断 {len(raw_obs)-550} 字符，如需详情请加过滤条件]...\n{tail}"

    async def run_task(self, user_query: str):
        context = [{"role": "user", "content": user_query}]
        consecutive_errors = 0

        for step in range(1, 9):  # 最大允许 8 步本地闭环工具调用
            # 1. 默认使用本地 3B 模型 + lora-executor 极速推理 (耗时 ~250ms)
            raw_decision = await self.local_llm.generate(
                messages=context,
                lora_adapter="lora-tool-executor",
                gbnf_grammar="agent_action.gbnf"
            )
            action = json.loads(raw_decision)
            self._append_event("local_action", action)

            # 2. 检查是否完成或主动请求云端支援
            if action["call"] == "final_answer":
                return action["args"]["reply"]

            if action["call"] == "escalate_to_cloud" or consecutive_errors >= 2:
                return await self._fallback_to_cloud(context, failed_action=action)

            # 3. 执行本地工具并清洗 Observation
            try:
                raw_result = await self.execute_local_tool(action["call"], action["args"])
                clean_obs = self.prune_observation(str(raw_result))
                context.append({"role": "assistant", "content": raw_decision})
                context.append({"role": "tool", "content": clean_obs})
                consecutive_errors = 0
            except Exception as err:
                consecutive_errors += 1
                context.append({"role": "assistant", "content": raw_decision})
                context.append({"role": "tool", "content": f"ERROR: {str(err)}"})

        return await self._fallback_to_cloud(context, failed_action={"reason": "max_steps_exceeded"})

    async def _fallback_to_cloud(self, context: list[dict], failed_action: dict):
        """云端大模型救场，并自动将这次救场记录为夜间增量微调的正样本（数据飞轮）"""
        cloud_response = await self.cloud_llm.solve_with_full_schema(context)
        with self.flywheel_log.open("a", encoding="utf-8") as f:
            f.write(json.dumps({
                "timestamp": time.time(),
                "context": context,
                "rejected_local": failed_action,
                "chosen_cloud": cloud_response
            }, ensure_ascii=False) + "\n")
        return cloud_response

    async def compact_sliding_window_if_needed(self, uncompacted_events: list[dict]):
        """后台静默切换至 lora-compactor，将事件流压缩为 Markdown 记忆补丁"""
        patch_json = await self.local_llm.generate(
            messages=[{"role": "user", "content": json.dumps(uncompacted_events, ensure_ascii=False)}],
            lora_adapter="lora-compactor"
        )
        return json.loads(patch_json)

    def _append_event(self, event_type: str, payload: dict):
        with self.event_log.open("a", encoding="utf-8") as f:
            f.write(json.dumps({"ts": time.time(), "type": event_type, "data": payload}, ensure_ascii=False) + "\n")
